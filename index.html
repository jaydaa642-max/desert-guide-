let deviceHeading = null; // اتجاه التفاف الجهاز بالدرجات (0 - 360) function handleOrientation(event) { let heading = null; // في أجهزة iOS/iPadOS يظهر الاتجاه الحقيقي في webkitCompassHeading if (typeof event.webkitCompassHeading === "number" && !Number.isNaN(event.webkitCompassHeading)) { heading = event.webkitCompassHeading; } else if (typeof event.alpha === "number" && event.absolute) { heading = (360 - event.alpha) % 360; } if (heading !== null) { deviceHeading = (heading + 360) % 360; updateNavigationUI(); } } function enableCompassIfNeeded() { if (typeof DeviceOrientationEvent !== "undefined" && typeof DeviceOrientationEvent.requestPermission === "function") { DeviceOrientationEvent.requestPermission() .then((state) => { if (state === "granted") { window.addEventListener("deviceorientation", handleOrientation, true); } }) .catch(() => {}); } else { window.addEventListener("deviceorientation", handleOrientation, true); } } // نختار اتجاه التفاف من البوصلة أولًا، وإلا من حركة GPS إن توفرت function getCurrentUserHeading() { if (typeof deviceHeading === "number" && !Number.isNaN(deviceHeading)) { return deviceHeading; } if (currentPos && typeof currentPos.heading === "number" && !Number.isNaN(currentPos.heading)) { return (currentPos.heading + 360) % 360; } return 0; } // رسم الرادار: يدور مع التفاف الشخص ويعرض النقطة فقط بدون خط أثر function drawRadarDotOnly(relativeBearingDeg, distMeters, userHeadingDeg) { const w = radarCanvas.width; const h = radarCanvas.height; const cx = w / 2; const cy = h / 2; const maxR = (w / 2) - 14; rCtx.clearRect(0, 0, w, h); // دوائر الرادار rCtx.strokeStyle = "#1e3a8a"; rCtx.lineWidth = 1; [0.35, 0.7, 1].forEach((ratio) => { rCtx.beginPath(); rCtx.arc(cx, cy, maxR * ratio, 0, Math.PI * 2); rCtx.stroke(); }); // المحاور الأربعة rCtx.beginPath(); rCtx.moveTo(cx - maxR, cy); rCtx.lineTo(cx + maxR, cy); rCtx.moveTo(cx, cy - maxR); rCtx.lineTo(cx, cy + maxR); rCtx.stroke(); // تدوير الحروف الأساسية عكس التفاف الجهاز ليبقى اتجاه الشمال صحيحًا بالنسبة للشخص const cardinals = [ { label: "ش", deg: 0 }, { label: "ق", deg: 90 }, { label: "ج", deg: 180 }, { label: "غ", deg: 270 } ]; rCtx.fillStyle = "#94a3b8"; rCtx.font = "bold 11px sans-serif"; rCtx.textAlign = "center"; rCtx.textBaseline = "middle"; cardinals.forEach((c) => { const angle = toRad(c.deg - userHeadingDeg); const lx = cx + Math.sin(angle) * (maxR - 4); const ly = cy - Math.cos(angle) * (maxR - 4); rCtx.fillText(c.label, lx, ly); }); // موقع النقطة بعد طرح التفاف الشخص لتعكس اتجاه الوجهة بالنسبة لدورانه const normalizedDist = Math.min(distMeters / 2000, 1); const rDot = normalizedDist * (maxR - 8); const rad = toRad(relativeBearingDeg); const dotX = cx + Math.sin(rad) * rDot; const dotY = cy - Math.cos(rad) * rDot; // النقطة فقط (بدون خط أثر) rCtx.fillStyle = "#22c55e"; rCtx.beginPath(); rCtx.arc(dotX, dotY, 6.5, 0, Math.PI * 2); rCtx.fill(); rCtx.strokeStyle = "#ffffff"; rCtx.lineWidth = 1.8; rCtx.stroke(); } function updateNavigationUI() { if (!currentPos || !activeTarget) return; const distM = distanceMeters(currentPos.lat, currentPos.lng, activeTarget.lat, activeTarget.lng); const bearing = calculateBearing(currentPos.lat, currentPos.lng, activeTarget.lat, activeTarget.lng); const userHeading = getCurrentUserHeading(); const relativeBearing = ((bearing - userHeading) % 360 + 360) % 360; const roundedBearing = Math.round(bearing); const roundedRelative = Math.round(relativeBearing); navPanelEl.classList.add("show"); directionBadgeEl.textContent = `الاتجاه: ${getCardinalArabic(bearing)}`; // السهم يدور حسب الاتجاه النسبي لدوران الشخص لتوضيح الطريق أمامه arrowIconEl.style.transform = `rotate(${roundedRelative}deg)`; targetNameEl.textContent = activeTarget.name; targetNoteEl.textContent = activeTarget.note || ""; if (distM < 1000) { distanceValEl.textContent = `المسافة المباشرة: ${Math.round(distM)} م`; } else { distanceValEl.textContent = `المسافة المباشرة: ${(distM / 1000).toFixed(2)} كم`; } bearingValEl.textContent = `زاوية الاتجاه: ${roundedBearing}° (النسبي: ${roundedRelative}°)`; drawRadarDotOnly(relativeBearing, distM, userHeading); if (distM <= 30) { alertBannerEl.classList.add("show"); if (!arrivalAlertShown) { arrivalAlertShown = true; playArrivalBeep(); } } else { alertBannerEl.classList.remove("show"); arrivalAlertShown = false; } }
